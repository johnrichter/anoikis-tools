func summarizeString(t *testing.T, input string) Summary {
	t.Helper()
	s, err := Summarize(strings.NewReader(input))
	if err != nil {
		t.Fatalf("Summarize returned unexpected error: %v", err)
	}
	return s
}

func render(s Summary) string {
	var buf bytes.Buffer
	Render(&buf, s)
	return buf.String()
}

func TestSummarize_AllFiveEventValues(t *testing.T) {
	input := `{"node_id":"n1","event":"dispatched","role":"r1"}
{"node_id":"n1","event":"complete","role":"r1"}
{"node_id":"n1","event":"failed","role":"r1"}
{"node_id":"n1","event":"merged","role":"r1"}
{"node_id":"n1","event":"grafted","role":"r1"}
`
	s := summarizeString(t, input)

	if s.EventsCounted != 5 || s.LinesSkipped != 0 {
		t.Fatalf("EventsCounted=%d LinesSkipped=%d, want 5 and 0", s.EventsCounted, s.LinesSkipped)
	}
	c := s.NodeCounts["n1"]
	if c == nil || c.Dispatched != 1 || c.Complete != 1 || c.Failed != 1 || c.Merged != 1 {
		t.Fatalf("unexpected node counts: %+v", c)
	}
	if got := s.EventKindCounts["grafted"]; got != 1 {
		t.Fatalf("EventKindCounts[grafted]=%d, want 1", got)
	}
	// grafted is not one of the four counted columns.
	if sum := c.Dispatched + c.Complete + c.Failed + c.Merged; sum != 4 {
		t.Fatalf("node column sum=%d, want 4 (grafted excluded)", sum)
	}
}

func TestSummarize_EmptyInput(t *testing.T) {
	s := summarizeString(t, "")
	if s.LinesRead != 0 || s.EventsCounted != 0 || s.LinesSkipped != 0 {
		t.Fatalf("expected all-zero counts for empty input, got %+v", s)
	}
	out := render(s)
	if !strings.Contains(out, "=== Per-node_id ===") || !strings.Contains(out, "=== Accounting ===") {
		t.Fatalf("expected full block structure even for empty input, got:\n%s", out)
	}
}

func TestSummarize_LineLongerThanDefaultScannerLimit(t *testing.T) {
	// bufio.Scanner's default token limit is 64KiB (bufio.MaxScanTokenSize).
	// Build a line comfortably over that, inside an ignored field, to prove
	// the raised buffer (design risk 4) prevents a spurious skip.
	padding := strings.Repeat("x", 70*1024)
	line := `{"node_id":"n1","event":"dispatched","role":"r1","detail":"` + padding + `"}`
	if len(line) <= 64*1024 {
		t.Fatalf("test line is not actually longer than 64KiB: %d bytes", len(line))
	}

	s := summarizeString(t, line+"\n")
	if s.LinesSkipped != 0 {
		t.Fatalf("LinesSkipped=%d, want 0 (long line must not be skipped)", s.LinesSkipped)
	}
	if s.EventsCounted != 1 {
		t.Fatalf("EventsCounted=%d, want 1", s.EventsCounted)
	}
}

	data, err := os.ReadFile("../../testdata/fixture.jsonl")
	if err != nil {
		t.Fatalf("reading fixture: %v", err)
	}

	if na == nil || na.Dispatched != 1 || na.Complete != 1 || na.Failed != 1 || na.Merged != 0 {
		t.Fatalf("node-a counts = %+v, want 1/1/1/0", na)
	}
	nb := s.NodeCounts["node-b"]
	if nb == nil || nb.Dispatched != 1 || nb.Complete != 1 || nb.Failed != 0 || nb.Merged != 1 {
		t.Fatalf("node-b counts = %+v, want 1/1/0/1", nb)
	}

func TestFixture_AddingUnpricedEventLeavesTotalByteIdentical(t *testing.T) {
	base, err := os.ReadFile("../../testdata/fixture.jsonl")
	if err != nil {
		t.Fatalf("reading fixture: %v", err)
	}

	before := render(summarizeString(t, string(base)))
	beforeTotal := extractLine(t, before, "total_cost_usd:")

	withNoUsage := string(base) + `{"node_id":"node-c","event":"dispatched","usage":{"known":false}}` + "\n"
	afterKnownFalse := render(summarizeString(t, withNoUsage))
	afterKnownFalseTotal := extractLine(t, afterKnownFalse, "total_cost_usd:")

	if beforeTotal != afterKnownFalseTotal {
		t.Fatalf("total changed after adding usage.known==false event: %q -> %q", beforeTotal, afterKnownFalseTotal)
	}

	withoutUsage := string(base) + `{"node_id":"node-c","event":"dispatched"}` + "\n"
	afterNoUsage := render(summarizeString(t, withoutUsage))
	afterNoUsageTotal := extractLine(t, afterNoUsage, "total_cost_usd:")

	if beforeTotal != afterNoUsageTotal {
		t.Fatalf("total changed after adding event with no usage object: %q -> %q", beforeTotal, afterNoUsageTotal)
	}
}

func extractLine(t *testing.T, block, prefix string) string {
	t.Helper()
	for _, line := range strings.Split(block, "\n") {
		if strings.HasPrefix(line, prefix) {
			return line
		}
	}
	t.Fatalf("no line with prefix %q found in:\n%s", prefix, block)
	return ""
}

func TestRender_SortOrderAscendingByByteOrder(t *testing.T) {
	input := `{"node_id":"z-node","event":"dispatched","role":"z-role"}
{"node_id":"a-node","event":"dispatched","role":"a-role"}
{"node_id":"m-node","event":"dispatched","role":"m-role"}
`
	out := render(summarizeString(t, input))

	nodeSection := sectionBetween(t, out, "=== Per-node_id ===", "=== Per-role ===")
	if got, want := indexOrder(nodeSection, []string{"a-node", "m-node", "z-node"}), true; got != want {
		t.Fatalf("node_id rows not in ascending byte order:\n%s", nodeSection)
	}

	roleSection := sectionBetween(t, out, "=== Per-role ===", "=== Cost ===")
	if got, want := indexOrder(roleSection, []string{"a-role", "m-role", "z-role"}), true; got != want {
		t.Fatalf("role rows not in ascending byte order:\n%s", roleSection)
	}
}

func sectionBetween(t *testing.T, block, start, end string) string {
	t.Helper()
	i := strings.Index(block, start)
	if i < 0 {
		t.Fatalf("section start %q not found", start)
	}
	j := strings.Index(block[i:], end)
	if j < 0 {
		t.Fatalf("section end %q not found", end)
	}
	return block[i : i+j]
}

func indexOrder(block string, wantAscending []string) bool {
	last := -1
	for _, want := range wantAscending {
		idx := strings.Index(block, want)
		if idx < 0 || idx < last {
			return false
		}
		last = idx
	}
	return true
}

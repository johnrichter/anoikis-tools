// Package runlogstats parses an anoikis run-log.jsonl stream and renders a
// deterministic plain-text summary of per-node_id and per-role event counts
// plus known-and-priced cost.
package runlogstats

import (
	"bufio"
	"encoding/json"
	"fmt"
	"io"
	"sort"
)

// noRoleLabel is used for events that carry no role. It cannot collide with a
// schema-valid role token, which must start with an alphanumeric character.
const noRoleLabel = "(none)"

// maxLineSize raises bufio.Scanner's default 64KiB token limit so a long
// event line (e.g. a large free-text detail field) is not spuriously skipped.
const maxLineSize = 10 * 1024 * 1024

// Counts holds the four event kinds this tool tracks per node_id or per role.
type Counts struct {
	Dispatched int
	Complete   int
	Failed     int
	Merged     int
}

// add increments the field matching event. Callers must only invoke this
// for a value already known to be one of the four counted kinds.
func (c *Counts) add(event string) {
	switch event {
	case "dispatched":
		c.Dispatched++
	case "complete":
		c.Complete++
	case "failed":
		c.Failed++
	case "merged":
		c.Merged++
	}
}

// Summary is the aggregated result of scanning one run-log.jsonl stream.
type Summary struct {
	NodeCounts map[string]*Counts
	RoleCounts map[string]*Counts

	// EventKindCounts breaks EventsCounted down by every event value seen,
	// including grafted and any value this tool does not otherwise count,
	// so schema drift shows up as a nonzero count rather than being dropped.
	EventKindCounts map[string]int

	TotalCost     float64
	UnpricedCount int

	LinesRead     int
	EventsCounted int
	LinesSkipped  int
}

type rawUsage struct {
	Known   *bool    `json:"known"`
	CostUSD *float64 `json:"cost_usd"`
}

type rawEvent struct {
	NodeID *string   `json:"node_id"`
	Event  *string   `json:"event"`
	Role   *string   `json:"role"`
	Usage  *rawUsage `json:"usage"`
}

func isCountedEvent(event string) bool {
	switch event {
	case "dispatched", "complete", "failed", "merged":
		return true
	default:
		return false
	}
}

func bucket(m map[string]*Counts, key string) *Counts {
	c, ok := m[key]
	if !ok {
		c = &Counts{}
		m[key] = c
	}
	return c
}

// Summarize reads one JSON object per line from r and returns the aggregated
// Summary. A line that is not valid JSON, or that lacks node_id or event, is
// skipped and tallied rather than treated as fatal. Summarize returns a
// non-nil error only when the underlying reader itself fails (io.Reader
// returned an error), since a read failure means the input could not be read
// at all rather than that one line was malformed.
func Summarize(r io.Reader) (Summary, error) {
	s := Summary{
		NodeCounts:      make(map[string]*Counts),
		RoleCounts:      make(map[string]*Counts),
		EventKindCounts: make(map[string]int),
	}

	scanner := bufio.NewScanner(r)
	scanner.Buffer(make([]byte, 0, 64*1024), maxLineSize)

	for scanner.Scan() {
		s.LinesRead++

		var raw rawEvent
		if err := json.Unmarshal(scanner.Bytes(), &raw); err != nil {
			s.LinesSkipped++
			continue
		}
		if raw.NodeID == nil || raw.Event == nil {
			s.LinesSkipped++
			continue
		}

		s.EventsCounted++
		event := *raw.Event
		s.EventKindCounts[event]++

		if isCountedEvent(event) {
			bucket(s.NodeCounts, *raw.NodeID).add(event)

			role := noRoleLabel
			if raw.Role != nil {
				role = *raw.Role
			}
			bucket(s.RoleCounts, role).add(event)
		}

		if raw.Usage != nil && raw.Usage.Known != nil && *raw.Usage.Known && raw.Usage.CostUSD != nil {
			s.TotalCost += *raw.Usage.CostUSD
		} else {
			s.UnpricedCount++
		}
	}
	if err := scanner.Err(); err != nil {
		return s, fmt.Errorf("reading input: %w", err)
	}

	return s, nil
}

func sortedKeys[V any](m map[string]V) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys
}

// Render writes s to w as one deterministic plain-text block. Rows are
// sorted ascending by byte order (node_id, role, event kind), never by Go
// map iteration order, so repeated calls on the same Summary produce
// byte-identical output.
func Render(w io.Writer, s Summary) {
	fmt.Fprintln(w, "=== Per-node_id ===")
	fmt.Fprintln(w, "node_id\tdispatched\tcomplete\tfailed\tmerged")
	for _, id := range sortedKeys(s.NodeCounts) {
		c := s.NodeCounts[id]
		fmt.Fprintf(w, "%s\t%d\t%d\t%d\t%d\n", id, c.Dispatched, c.Complete, c.Failed, c.Merged)
	}
	fmt.Fprintln(w)

	fmt.Fprintln(w, "=== Per-role ===")
	fmt.Fprintln(w, "role\tdispatched\tcomplete\tfailed\tmerged")
	for _, role := range sortedKeys(s.RoleCounts) {
		c := s.RoleCounts[role]
		fmt.Fprintf(w, "%s\t%d\t%d\t%d\t%d\n", role, c.Dispatched, c.Complete, c.Failed, c.Merged)
	}
	fmt.Fprintln(w)

	fmt.Fprintln(w, "=== Cost ===")
	fmt.Fprintf(w, "total_cost_usd: %.4f\n", s.TotalCost)
	fmt.Fprintf(w, "unpriced_events: %d\n", s.UnpricedCount)
	fmt.Fprintln(w, "note: only usage.known == true events with cost_usd present are summed; all other events (no usage, known == false, known == true without cost_usd) count as unpriced, never as 0")
	fmt.Fprintln(w)

	fmt.Fprintln(w, "=== Accounting ===")
	fmt.Fprintf(w, "lines_read: %d\n", s.LinesRead)
	fmt.Fprintf(w, "events_counted: %d\n", s.EventsCounted)
	fmt.Fprintf(w, "lines_skipped: %d\n", s.LinesSkipped)
	fmt.Fprintln(w, "note: lines_read = events_counted + lines_skipped")
	fmt.Fprint(w, "events_counted_by_kind:")
	for _, kind := range sortedKeys(s.EventKindCounts) {
		fmt.Fprintf(w, " %s=%d", kind, s.EventKindCounts[kind])
	}
	fmt.Fprintln(w)
	fmt.Fprintln(w, "note: events_counted = sum of events_counted_by_kind values")
}
